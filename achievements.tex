\section{Soft Skills}

\begin{itemize}
    \item \textbf{Teamwork}: Effective collaboration with cross-functional teams
    \item \textbf{Communication}: Clear transmission of technical concepts to stakeholders
    \item \textbf{Technical Leadership}: Ability to guide architectural decisions
    \item \textbf{Problem Solving}: Analytical approach to identify and resolve complex challenges
    \item \textbf{Organization}: Rigorous management of priorities and deadline adherence
    \item \textbf{Adaptability}: Quick learning of new technologies and methodologies
\end{itemize}
